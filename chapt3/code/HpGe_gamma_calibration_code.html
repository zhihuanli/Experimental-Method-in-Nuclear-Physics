HPGe $\gamma$ 能谱刻度：ROOT 实例¶

本页使用同目录下的 gamma.root 演示 peak-shape comparison、能量刻度及其残差、FWHM–$E$ 和 apparent full-energy peak efficiency。代码采用常规 ROOT fit：histogram 用 TH1::Fit(TF1)，刻度曲线用 TGraphErrors::Fit(TF1)。

在页面开头选择 Python / PyROOT 或 ROOT C++。两个版本读取同一数据并执行相同分析；页面显示的结果由 ROOT C++ 版本生成。

Peak fit 与本底模型¶

局部平滑本底先用 Gaussian + linear background。若低能侧本底在峰附近形成明显台阶，加入

$$\frac{S}{2}\operatorname{erfc}\!\left(\frac{x-\mu}{\sqrt{2}\sigma}\right).$$

这个 erfc 项描述 photopeak 低能侧的 step；它不等同于任意提高多项式阶数。下面在 1408 keV 峰附近比较两种模型，并画出 Pearson residual $(n_i-\mu_i)/\sqrt{\mu_i}$。本数据统计量很高，因此很小的峰形偏差也会形成明显残差。加入 step 后残差减小但没有完全消失，说明真实峰形仍比这一教学模型复杂。

随后用含 step 的模型提取全部刻度峰。实际分析中应逐峰查看残差；step 参数与零相容且残差没有结构时，保留更简单的本底模型。

def fit_peak(hist, name, guess, width, use_step):
    xmin, xmax = guess - width, guess + width
    left = hist.GetBinContent(hist.FindBin(xmin))
    right = hist.GetBinContent(hist.FindBin(xmax))
    slope = (right - left) / (xmax - xmin)
    intercept = left - slope * xmin
    height = max(hist.GetBinContent(hist.FindBin(guess))
                 - intercept - slope * guess, 1.0)

    formula = "gaus(0)+pol1(3)"
    if use_step:
        formula += "+[5]*0.5*TMath::Erfc((x-[1])/(sqrt(2)*[2]))"
    function = ROOT.TF1(name, formula, xmin, xmax)
    function.SetParameters(height, guess, 0.8, intercept, slope)
    function.SetParLimits(0, 0.0, 10.0 * height)
    function.SetParLimits(1, guess - 2.0, guess + 2.0)
    function.SetParLimits(2, 0.2, 3.0)
    if use_step:
        function.SetParameter(5, max(left - right, 0.0))
        function.SetParLimits(5, 0.0, height)

    # L: binned Poisson likelihood, I: integrate over each bin,
    # R: TF1 range, S: return fit result, Q: quiet, N: do not attach/draw.
    result = hist.Fit(function, "LIRSQN")

    amplitude = function.GetParameter(0)
    sigma = abs(function.GetParameter(2))
    bin_width = hist.GetBinWidth(1)
    area_factor = math.sqrt(2.0 * math.pi) / bin_width
    area = amplitude * sigma * area_factor

    # Propagate the covariance of Gaussian height and sigma to the area.
    covariance = result.GetCovarianceMatrix()
    d_area_d_a = sigma * area_factor
    d_area_d_sigma = amplitude * area_factor
    area_variance = (
        d_area_d_a**2 * covariance[0][0]
        + d_area_d_sigma**2 * covariance[2][2]
        + 2.0 * d_area_d_a * d_area_d_sigma * covariance[0][2]
    )
    return {
        "function": function,
        "status": int(result),
        "mean": function.GetParameter(1),
        "mean_error": function.GetParError(1),
        "sigma": sigma,
        "sigma_error": function.GetParError(2),
        "area": area,
        "area_error": math.sqrt(max(area_variance, 0.0)),
    }

def residual_graph(hist, function, xmin, xmax, name):
    graph = ROOT.TGraph()
    graph.SetName(name)
    first_bin = hist.FindBin(xmin)
    last_bin = hist.FindBin(xmax)
    point = 0
    for bin_number in range(first_bin, last_bin + 1):
        low_edge = hist.GetBinLowEdge(bin_number)
        width = hist.GetBinWidth(bin_number)
        expected = function.Integral(low_edge, low_edge + width) / width
        observed = hist.GetBinContent(bin_number)
        if expected > 0.0:
            residual = (observed - expected) / math.sqrt(expected)
            graph.SetPoint(point, hist.GetBinCenter(bin_number), residual)
            point += 1
    return graph

struct PeakMeasurement {
    TF1* function;
    int status;
    double mean;
    double meanError;
    double sigma;
    double sigmaError;
    double area;
    double areaError;
};

auto fitPeak = [&](TH1* hist, const std::string& name,
                   double guess, double width, bool useStep) -> PeakMeasurement {
    const double xmin = guess - width;
    const double xmax = guess + width;
    const double left = hist->GetBinContent(hist->FindBin(xmin));
    const double right = hist->GetBinContent(hist->FindBin(xmax));
    const double slope = (right - left) / (xmax - xmin);
    const double intercept = left - slope * xmin;
    const double height = std::max(
        hist->GetBinContent(hist->FindBin(guess)) - intercept - slope * guess, 1.0);

    std::string formula = "gaus(0)+pol1(3)";
    if (useStep) {
        formula += "+[5]*0.5*TMath::Erfc((x-[1])/(sqrt(2)*[2]))";
    }
    auto function = new TF1(name.c_str(), formula.c_str(), xmin, xmax);
    function->SetParameters(height, guess, 0.8, intercept, slope);
    function->SetParLimits(0, 0.0, 10.0 * height);
    function->SetParLimits(1, guess - 2.0, guess + 2.0);
    function->SetParLimits(2, 0.2, 3.0);
    if (useStep) {
        function->SetParameter(5, std::max(left - right, 0.0));
        function->SetParLimits(5, 0.0, height);
    }

    // L: binned Poisson likelihood, I: integrate over each bin,
    // R: TF1 range, S: return fit result, Q: quiet, N: do not attach/draw.
    TFitResultPtr result = hist->Fit(function, "LIRSQN");

    const double amplitude = function->GetParameter(0);
    const double sigma = std::abs(function->GetParameter(2));
    const double binWidth = hist->GetBinWidth(1);
    const double areaFactor = std::sqrt(2.0 * TMath::Pi()) / binWidth;
    const double area = amplitude * sigma * areaFactor;

    // Propagate the covariance of Gaussian height and sigma to the area.
    const auto covariance = result->GetCovarianceMatrix();
    const double dAreaDA = sigma * areaFactor;
    const double dAreaDSigma = amplitude * areaFactor;
    const double areaVariance =
        dAreaDA * dAreaDA * covariance(0, 0)
        + dAreaDSigma * dAreaDSigma * covariance(2, 2)
        + 2.0 * dAreaDA * dAreaDSigma * covariance(0, 2);

    return {function, static_cast<int>(result), function->GetParameter(1),
            function->GetParError(1), sigma, function->GetParError(2),
            area, std::sqrt(std::max(areaVariance, 0.0))};
};

auto residualGraph = [&](TH1* hist, TF1* function, double xmin,
                         double xmax, const std::string& name) {
    auto graph = new TGraph();
    graph->SetName(name.c_str());
    int point = 0;
    for (int bin = hist->FindBin(xmin); bin <= hist->FindBin(xmax); ++bin) {
        const double lowEdge = hist->GetBinLowEdge(bin);
        const double width = hist->GetBinWidth(bin);
        const double expected = function->Integral(lowEdge, lowEdge + width) / width;
        const double observed = hist->GetBinContent(bin);
        if (expected > 0.0) {
            graph->SetPoint(point++, hist->GetBinCenter(bin),
                            (observed - expected) / std::sqrt(expected));
        }
    }
    return graph;
};

能量刻度与残差¶

峰中心误差放在 TGraphErrors 的横坐标误差中；参考能量的不确定度在本例尺度下忽略。选项 F 强制使用通用 minimizer，使拟合考虑横坐标误差。一次和二次函数都使用相同的峰中心；真正决定函数阶数的是下面的残差图。

对这组数据，二次项只带来很小的整体变化，并未改善最大残差，因此后续示例采用一次刻度。峰中心的统计误差小于实际残差，说明 peak model 和 calibration model 的系统效应已经占主导；不应人为放大误差来强迫 $\chi^2/\mathrm{ndf}$ 接近 1。

FWHM–$E$ 曲线¶

这里将各峰的 $\sigma_{ch}$ 用所选刻度曲线的局部导数换算为 $\sigma_E$，再计算 FWHM。曲线使用

$$FWHM(E)=\sqrt{a^2+b^2E+c^2E^2}$$

以保证峰宽为正。图中误差只传播了 peak fit 对 $\sigma$ 的统计误差；刻度函数和峰形模型的系统效应没有包含，因此残差比误差棒大是预期的诊断结果。

Apparent full-energy peak efficiency¶

Gaussian signal 的面积由 height、$\sigma$ 和二者 covariance 计算。示例用参考活度、半衰期、gamma emission probability 和 7442 s 计算 efficiency，并用二次 $\ln\varepsilon$–$\ln E$ 函数描述趋势。

图中误差棒包含 peak-area、gamma emission probability 和源活度误差。源活度误差对同一核素的所有点相关，因此曲线只作为经验趋势；不能把这里普通 TGraphErrors::Fit 的 $\chi^2$ 当作严格 goodness-of-fit。数据未提供 dead-time 和 true-coincidence-summing correction，故纵轴明确标为 apparent efficiency。

结果的使用范围¶

  • 对当前 histogram，常规 ROOT fit 已足够；RooFit 适合以后需要 event-by-event likelihood、多峰共享参数、多个谱同时拟合或显式 nuisance parameters 的分析。
  • peak-fit residual、calibration residual、FWHM residual 和 efficiency residual 分别检验不同层次的问题，不能互相替代。
  • 本页得到的 efficiency 尚未包含 dead time、true-coincidence summing、源几何与自吸收修正，只能用于理解完整效率刻度流程和比较能量依赖。

峰能量、half-life 和 gamma emission probability 应以标准源证书和同一套 evaluated data 为准，例如 DDEP/LNHB nuclear data。ROOT 的 L、I、R、S 等拟合选项参见 ROOT histogram fitting manual。
